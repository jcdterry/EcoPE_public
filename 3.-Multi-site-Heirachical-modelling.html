Introduction

This document demonstrates the ability for the heierarchical (multi-site) regression model to identify trends from short simulated data.

Without known values, it isn’t really meaningful to conduct a formal power analysis. To demonstrate the abiltiy to gain increasing information from more sites, I fit the model to short (52 years = 50 motifs) data from increasing numbers of sites.

Simulated data is generated from the same Ricker based model, with two sources of noise - global and site-specific.

Data

Data bundling.

For the STAN function data needs to be in list form:

  • N = nrow(event_data),
  • K = length(motif_labels_3),
  • S = max(site_index),
  • G = max(time_index),
  • y = as.integer(event_data$motif_id),
  • site = site_index,
  • time_id = time_index,
  • time_scaled = event_data$time_scaled,
  • time_grid_scaled = time_lookup$time_scaled,
  • prior_only = as.integer(prior_only),
  • slope_prior_sd = slope_prior_sd,
  • sigma_site_scale = sigma_site_scale,
  • sigma_period_scale = sigma_period_scale
make_multisite_linear_pe_trend_bundle <- function(simulated_data_matrix,
                                                  m = 3, #  motif length
                                                  site_ids = NULL,
                                                  prior_only = 0,
                                                  slope_prior_sd = 0.5,
                                                  sigma_site_scale = 0.35,
                                                  sigma_period_scale = 0.15) {
  
  abundance <- simulated_data_matrix
  if (is.null(colnames(abundance))) {colnames(abundance) <- paste0("site_", seq_len(ncol(abundance)))}
  if (is.null(site_ids)) {site_ids <- seq_len(ncol(abundance))}
  if (any(site_ids < 1 | site_ids > ncol(abundance))) {
    stop("site_ids must index columns of the abundance matrix.", call. = FALSE)}
  
  ### Calculate Motifs
  
  k <- as.integer(factorial(m))
  n_year <- nrow(abundance)
  center_time <- seq.int(2L, n_year - 1L) # center time of motifs
  x_range <- range(center_time, na.rm = TRUE)
  time_grid_scaled <-(center_time - mean(x_range)) / diff(x_range)
  
  rows <- vector("list", length(site_ids))
  
  event_data <- map_df(seq_along(site_ids),function(i){
    site_col <- site_ids[i]
    motif_id <- statcomp::ordinal_pattern_time_series(abundance[, site_col], m) + 1L
    motif_id <- motif_id[seq_len(length(center_time))]
    
    if (any(is.na(motif_id))) {keep <- !is.na(motif_id)} else {keep <- rep(TRUE, length(motif_id))}
    
    data.frame(site_id = i,
               site_column = site_col,
               site_name = colnames(abundance)[site_col],
               center_time = center_time[keep],
               time_id = seq_along(center_time)[keep],
               time_scaled = time_grid_scaled[keep],
               motif_id = as.integer(motif_id[keep]))
  })
  
  row.names(event_data) <- NULL
  
  ##   Build List
  stan_data <- list(N = nrow(event_data),
                    K = k,
                    S = length(site_ids),
                    G = length(time_grid_scaled),
                    y = as.integer(event_data$motif_id),
                    site = as.integer(event_data$site_id),
                    time_id = as.integer(event_data$time_id),
                    time_scaled = as.numeric(event_data$time_scaled),
                    time_grid_scaled = as.numeric(time_grid_scaled),
                    #Prior Flag 
                    prior_only = as.integer(prior_only),
                    ## Changeable Priors
                    slope_prior_sd = slope_prior_sd,
                    sigma_site_scale = sigma_site_scale,
                    sigma_period_scale = sigma_period_scale)
  
  ## Checkers
  stopifnot( length(stan_data$time_scaled) == stan_data$N,
             length(stan_data$time_grid_scaled) == stan_data$G,
             length(stan_data$y) == stan_data$N,
             length(stan_data$site) == stan_data$N,
             length(stan_data$time_id) == stan_data$N)
  
  return(stan_data)
}

Different Numbers of sites.

n_site_grid <- c(1,seq(from=5,to=65, by 5))

NSiteTest_df <- map_df(n_site_grid, function(n_site) {
  ## Simulate raw data
  data_x <-   simulate_feedback_sites(n_site = n_site,
                                      n_year = 52,
                                      delayed_feedback = -1.1, 
                                      rho_spatial = 0.25,
                                      sd_start = 0.1, sd_end = 0.2)
  ## Bundle Data
  multi_site_bundle <- make_multisite_linear_pe_trend_bundle(simulated_data_matrix = data_x$abundance_matrix,
                                                             slope_prior_sd = 0.5,
                                                             sigma_site_scale = 0.35,
                                                             sigma_period_scale = 0.15)
  ## Fit Model
  multi_site_model_fit <- multi_site_model$sample(data = multi_site_bundle,
                                                  chains = 4, parallel_chains = 4,
                                                  iter_warmup = 500,iter_sampling = 800,
                                                  refresh = 100)
  ## Prep Posterior
  multi_site_summary <- multi_site_model_fit$summary(c(  "pe_slope",  "pe_intercept",
                                                         "pe_population_time","pe_period_time",
                                                         "alpha_nsb", "beta_time",
                                                         "sigma_site","sigma_period"))
  ## Label and return
  multi_site_summary$n_site = n_site
  return(multi_site_summary)
})
